<!-- JavaScript for WhatsApp Integration -->
<script>
const BUSINESS_PHONE_NUMBER = "254715916405";
const MPESA_POCHI_NUMBER = "0715916405";
const OPENING_HOUR = 9; // 9:00 AM
const CLOSING_HOUR = 20; // 8:00 PM (20:00)

function isBusinessOpen() {
const now = new Date();
const currentHour = now.getHours();
return currentHour >= OPENING_HOUR && currentHour < CLOSING_HOUR;
}

function processWhatsAppCheckout() {
// Retrieve Selected Form Values
const packSizeElement = document.getElementById("packSize");
const quantityElement = document.getElementById("quantity");
const locationElement = document.getElementById("deliveryLocation");
const customerNameElement = document.getElementById("customerName");

const flavorCheckboxes = document.querySelectorAll('input[name="flavors"]:checked');
const selectedFlavors = Array.from(flavorCheckboxes).map(cb => cb.value);

// Validation
if (!customerNameElement || !customerNameElement.value.trim()) {
alert("Please enter your name before placing the order.");
customerNameElement.focus();
return;
}

if (selectedFlavors.length === 0) {
alert("Please select at least one flavor mix.");
return;
}

// Check operating hours
if (!isBusinessOpen()) {
const proceed = confirm(
"Our kitchen is currently closed (Operating Hours: 9:00 AM - 8:00 PM).\n\n" +
"We will process and dispatch your order first thing at 9:00 AM tomorrow!\n\n" +
"Proceed to WhatsApp to send the order?"
);
if (!proceed) return;
}

const packSize = packSizeElement ? packSizeElement.value : "Standard Pack";
const quantity = quantityElement ? parseInt(quantityElement.value, 10) : 1;
const location = locationElement ? locationElement.value : "Standard Delivery";
const customerName = customerNameElement.value.trim();

// Price Lookup
const priceMap = {
"Small Pack (100g)": 100,
"Medium Pack (250g)": 220,
"Jumbo Pack (500g)": 400,
"Wholesale Box (2kg)": 1400
};
const deliveryMap = {
"Thika Town / Env.": 100,
"Nairobi CBD": 200,
"Juja / Ruiru": 150,
"Rest of Kenya (Courier)": 300
};

const pricePerPack = priceMap[packSize] || 100;
const subtotal = pricePerPack * quantity;
const deliveryFee = deliveryMap[location] || 150;
const grandTotal = subtotal + deliveryFee;

// Construct WhatsApp Message
const message = `*NEW MABUYU ORDER* 🛍️\n\n` +
`*Customer:* ${customerName}\n` +
`*Pack Size:* ${packSize}\n` +
`*Quantity:* ${quantity} pack(s)\n` +
`*Flavor Mix:* ${selectedFlavors.join(", ")}\n` +
`*Dispatch:* ${location}\n\n` +
`---------------------------\n` +
`*Subtotal:* KSh ${subtotal.toLocaleString()}\n` +
`*Est. Delivery:* KSh ${deliveryFee.toLocaleString()}\n` +
`*Total Amount:* KSh ${grandTotal.toLocaleString()}\n` +
`---------------------------\n\n` +
`💳 *Payment:* M-Pesa Pochi\n` +
`📱 *Pochi No:* ${MPESA_POCHI_NUMBER}\n\n` +
`Hello Saro! I would like to place this order. Please confirm availability.`;

const encodedMessage = encodeURIComponent(message);
const whatsappUrl = `https://wa.me/${BUSINESS_PHONE_NUMBER}?text=${encodedMessage}`;

window.open(whatsappUrl, "_blank");
}
</script>
